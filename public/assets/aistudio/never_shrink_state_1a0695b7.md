  2. **ترقية السيرفر تلقائياً من الجهاز الذي يملك العدد الأكبر (`Auto-Heal Server from Richest Client`):**
     - إذا كان متصفح اللابتوب يحمل في ذاكرته المحلية عدد أعضاء أكبر مما أرجعه السيرفر (مثلاً اللابتوب فيه 93-94 عضواً بينما السيرفر رجع 65 لأن AI Studio كتب فوقه)، يقوم اللابتوب تلقائياً بدمج القائمتين معاً ورفع القائمة المدمجة الكبرى (94+ عضواً) إلى `Cloudflare Database` لتستفيد منها كل الهواتف وAI Studio فوراً!
  3. **قاعدة `Never-Shrink State` الصارمة في `setMembers`:**
     ```javascript
     setMembers(prevMembers => {
       if (!incomingMembers || !Array.isArray(incomingMembers) || incomingMembers.length === 0) {
         return prevMembers;
       }
       // 1. مستحيل أن ينكمش الـ State من قائمة كبيرة إلى قائمة أصغر (مثل النزول من 94 إلى 65 أو إلى 1)!
       // بل ندمج دائماً القائمتين معاً (Union Merge) لنحافظ على الأعلى دائماً:
       const unionMap = new Map();
       // نضع الأعضاء الحاليين أولاً (بشرط ألا يكونوا مجرد عضو افتراضي وهمي بدون بيانات)
       prevMembers.forEach((m, idx) => {
         if (m && (m.id || m.name)) {
           const key = String(m.id || m.name).trim();
           unionMap.set(key, m);
         }
       });
       // ندمج الأعضاء القادمين من السيرفر بدون إسقاط أي صف
       incomingMembers.forEach((m, idx) => {
         if (m && (m.id || m.name)) {
           let key = String(m.id || `${m.name}-${idx}`).trim();
           const existing = unionMap.get(key);
           if (existing && normalizeArabic(existing.name) !== normalizeArabic(m.name)) {
             // إذا تصادف تشابه الـ ID لشخصين مختلفين في الاسم، لا تدمجهما! بل افصل المفتاح ليظهر الاثنان:
             key = `${key}-ROW-${idx + 1}`;
           }
           unionMap.set(key, {
             ...(existing || {}),
             ...m,
             id: key,
             originalId: m.id || existing?.id,
             status: 'active',
             isDeleted: false
           });
         }
       });
       const mergedList = Array.from(unionMap.values());
       if (mergedList.length > 5) {
         try { localStorage.setItem('SAFE_BACKUP_MEMBERS', JSON.stringify(mergedList)); } catch (e) {}
       }
       return mergedList;
     });
     ```
