       ```javascript
       export function getNormalizedGender(member): 'male' | 'female' {
         const raw = String(member?.gender || member?.sex || member?.type || '').trim().toLowerCase();
         if (
           raw.includes('انث') ||
           raw.includes('أنث') ||
           raw.includes('شابات') ||
           raw.includes('شابة') ||
           raw.includes('شابه') ||
           raw.includes('بنات') ||
           raw.includes('بنت') ||
           raw === 'female' ||
           raw === 'f'
         ) {
           return 'female';
         }
         return 'male'; // ذكر / شاب / شباب افتراضياً
       }
       ```
